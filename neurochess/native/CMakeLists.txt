cmake_minimum_required(VERSION 3.24)
project(NeuroChessNative LANGUAGES CXX CUDA)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native)
endif()
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)
find_path(TENSORRT_INCLUDE_DIR NvInferRuntime.h
  HINTS "${TENSORRT_ROOT}/include" REQUIRED)
find_library(TENSORRT_LIBRARY NAMES nvinfer libnvinfer.so.11
  HINTS "${TENSORRT_LIB_DIR}" "${TENSORRT_ROOT}/lib" REQUIRED)
find_library(TENSORRT_PLUGIN_LIBRARY NAMES nvinfer_plugin libnvinfer_plugin.so.11
  HINTS "${TENSORRT_LIB_DIR}" "${TENSORRT_ROOT}/lib" REQUIRED)
add_library(nc_inference SHARED trt_inference.cu)
target_include_directories(nc_inference SYSTEM PRIVATE "${TENSORRT_INCLUDE_DIR}")
target_link_libraries(nc_inference PRIVATE "${TENSORRT_LIBRARY}" "${TENSORRT_PLUGIN_LIBRARY}" CUDA::cudart)
target_compile_options(nc_inference PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
add_executable(neurochess_native native_player.cpp)
target_link_libraries(neurochess_native PRIVATE nc_inference Threads::Threads)
target_compile_options(neurochess_native PRIVATE -march=native)
install(TARGETS nc_inference neurochess_native RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
